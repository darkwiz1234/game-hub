<script>
const canvas = document.getElementById('gameCanvas');
const ctx = canvas.getContext('2d');
const uiProgressText = document.getElementById('progress-text');
const uiProgressBar = document.getElementById('progress-bar-fill');
const uiBest = document.getElementById('best');
const pauseBtn = document.getElementById('pauseBtn');
const resetBtn = document.getElementById('resetBtn');
const overlayActionBtn = document.getElementById('overlayActionBtn');
const msgOverlay = document.getElementById('message-overlay');
const msgText = document.getElementById('message-text');
const bgMusic = document.getElementById('sfx-music');
const hitSound = document.getElementById('sfx-collision');

let gameState = 'START';
let bestProgress = parseInt(localStorage.getItem('objectRunMap3Best') || 0);
uiBest.innerText = `BEST: ${bestProgress}%`;

const TILE_SIZE = 40;

/*
  LEGEND:
  '.' = Air
  '#' = Solid Block
  '^' = Floor Spike
  'v' = Ceiling Spike
  'P' = Ship Portal (Turns player into Triangle Ship)
  'C' = Cube Portal (Turns player back into Square Runner)
  'M' = Moving Obstacle (Oscillates vertically in Ship sections)
*/
const mapGrid = [
    "..................................................................................................................................................................................................................................................................................................................................................................................................................................................................................................................................................................",
    "................................................................................................................................................................vvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvvv............................................................................................................vvvvvvvvvvvv................................................................",
    "..................................................................................................................................................................................................................................................................................................................................................................................................................................................................",
    "....................................................................................................................................................................................M...................M...................M...................M....................................................................................................................................................................................",
    "..................................................................................................................................................................................................................................................................................................................................................................................................................................................................",
    "..................................................................................................................................................................................................................................................................................................................................................................................................................................................................",
    "...................................................................................................................................................................####...................................................................................####................................................................................................................................................................................",
    "........................................................................................................................####............................................................................####..................................................................C...................####........................P...................####..................................................................................",
    "......................................................................................####................................................................................................................................................................................................................................................................................................................................................................",
    "..................................#.......##....^......#.......^......##.......^^...................##...^^...##.......^....^^^^....#........^^..........^^^..........P...^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^^...^^^...#........^^^...##.........^^...#...##...###...^...^^^^^^^^^^^^...^^...##...^...##......................................",
    "##################################################################################################################################################################################################################################################################################################################################################################################################################################################################################################################################################################"
];

let blocks = [];
let spikes = [];
let portals = [];
let movingObstacles = [];
let obstacleXCoords = [];

function parseLevel() {
    blocks = []; spikes = []; portals = []; movingObstacles = []; obstacleXCoords = [];
    const recordedCols = new Set();

    for (let r = 0; r < mapGrid.length; r++) {
        for (let c = 0; c < mapGrid[r].length; c++) {
            let ch = mapGrid[r][c];
            let x = c * TILE_SIZE;
            let y = r * TILE_SIZE;

            if (ch === '#') {
                blocks.push({ x, y, w: TILE_SIZE, h: TILE_SIZE });
                if (r < 10 && !recordedCols.has(c)) { obstacleXCoords.push(x); recordedCols.add(c); }
            } else if (ch === '^') {
                spikes.push({ x, y, type: 'up' });
                if (!recordedCols.has(c)) { obstacleXCoords.push(x); recordedCols.add(c); }
            } else if (ch === 'v') {
                spikes.push({ x, y, type: 'down' });
                if (!recordedCols.has(c)) { obstacleXCoords.push(x); recordedCols.add(c); }
            } else if (ch === 'P') {
                portals.push({ x, y, toMode: 'SHIP' });
            } else if (ch === 'C') {
                portals.push({ x, y, toMode: 'CUBE' });
            } else if (ch === 'M') {
                movingObstacles.push({ x, baseY: y, y: y, size: 36, speed: 2.2, dir: 1, range: 60 });
                if (!recordedCols.has(c)) { obstacleXCoords.push(x); recordedCols.add(c); }
            }
        }
    }
    obstacleXCoords.sort((a, b) => a - b);
}
parseLevel();

let player = {
    x: 120, y: 320, size: 32,
    vx: 5.8, vy: 0,
    gravity: 0.75, jumpForce: -12.4,
    angle: 0, onGround: false,
    mode: 'CUBE' // 'CUBE' or 'SHIP'
};

let inputHeld = false;
let cameraX = 0;
let particles = [];
let trailParticles = [];
let portalBurstParticles = [];
let musicStarted = false;
let currentProgress = 0;
const totalObstacles = Math.max(1, obstacleXCoords.length);

function getTheme(pct) {
    if (pct >= 75) return { main: '#ff2a4b', glow: 'rgba(255, 42, 75, 0.4)', bg1: '#140306', bg2: '#2a050d' }; // High-tempo Crimson Red
    if (pct >= 50) return { main: '#b537f2', glow: 'rgba(181, 55, 242, 0.4)', bg1: '#0d0417', bg2: '#200833' }; // Ship Portal Purple
    return { main: '#00f0ff', glow: 'rgba(0, 240, 255, 0.4)', bg1: '#050614', bg2: '#0b162c' };                 // Classic Cyber Cyan
}

function resetGame() {
    player = {
        x: 120, y: 320, size: 32,
        vx: 5.8, vy: 0,
        gravity: 0.75, jumpForce: -12.4,
        angle: 0, onGround: false,
        mode: 'CUBE'
    };
    cameraX = 0;
    particles = [];
    trailParticles = [];
    portalBurstParticles = [];
    currentProgress = 0;
    inputHeld = false;
    gameState = 'PLAYING';
    msgOverlay.style.display = 'none';
    if (musicStarted) { bgMusic.currentTime = 0; bgMusic.play().catch(() => {}); }
}

function die() {
    gameState = 'GAMEOVER';
    hitSound.currentTime = 0; hitSound.play().catch(() => {});
    bgMusic.pause();
    msgText.innerText = "GAME OVER";
    msgText.style.color = "#ff0055";
    overlayActionBtn.innerText = "TRY AGAIN";
    msgOverlay.style.display = 'flex';

    for (let i = 0; i < 20; i++) {
        particles.push({
            x: player.x + player.size / 2, y: player.y + player.size / 2,
            vx: (Math.random() - 0.5) * 12, vy: (Math.random() - 1.1) * 11,
            size: Math.random() * 8 + 3, rot: Math.random() * Math.PI,
            rotSpeed: (Math.random() - 0.5) * 0.3, life: 1.0,
            color: i % 2 === 0 ? getTheme(currentProgress).main : '#fff'
        });
    }
}

function win() {
    gameState = 'WIN';
    msgText.innerText = "🎉 MAP 3 COMPLETE! 🎉";
    msgText.style.color = "#00f0ff";
    overlayActionBtn.innerText = "PLAY AGAIN";
    msgOverlay.style.display = 'flex';
}

function triggerJumpOrFly() {
    if (gameState === 'START') {
        musicStarted = true;
        bgMusic.play().catch(() => {});
        gameState = 'PLAYING';
    }
    if (gameState === 'PLAYING' && player.mode === 'CUBE' && player.onGround) {
        player.vy = player.jumpForce;
        player.onGround = false;
    }
}

window.addEventListener('keydown', (e) => {
    if (e.code === 'Space' || e.code === 'ArrowUp') {
        e.preventDefault();
        inputHeld = true;
        triggerJumpOrFly();
    }
});

window.addEventListener('keyup', (e) => {
    if (e.code === 'Space' || e.code === 'ArrowUp') inputHeld = false;
});

window.addEventListener('mousedown', (e) => {
    if (e.target.tagName === 'CANVAS') {
        inputHeld = true;
        triggerJumpOrFly();
    }
});

window.addEventListener('mouseup', () => { inputHeld = false; });
window.addEventListener('touchstart', (e) => {
    if (e.target.tagName === 'CANVAS') {
        inputHeld = true;
        triggerJumpOrFly();
    }
});
window.addEventListener('touchend', () => { inputHeld = false; });

function togglePause() {
    if (gameState === 'PLAYING') {
        gameState = 'PAUSED';
        bgMusic.pause();
        msgText.innerText = "PAUSED";
        msgText.style.color = "#ffffff";
        overlayActionBtn.innerText = "RESUME";
        msgOverlay.style.display = 'flex';
    } else if (gameState === 'PAUSED') {
        gameState = 'PLAYING';
        bgMusic.play().catch(() => {});
        msgOverlay.style.display = 'none';
    }
}

pauseBtn.addEventListener('click', togglePause);
resetBtn.addEventListener('click', resetGame);
overlayActionBtn.addEventListener('click', () => {
    if (gameState === 'PAUSED') togglePause();
    else resetGame();
});

function update() {
    if (gameState !== 'PLAYING') return;

    // Movement & Physics based on current mode
    if (player.mode === 'CUBE') {
        player.vy += player.gravity;
        if (!player.onGround) player.angle += 0.125;
        else player.angle = Math.round(player.angle / (Math.PI / 2)) * (Math.PI / 2);
    } else if (player.mode === 'SHIP') {
        // GD Ship Physics: Hold to fly, release to fall
        const shipUpwardThrust = -0.68;
        const shipGravity = 0.46;
        const maxVerticalSpeed = 6.8;

        if (inputHeld) {
            player.vy += shipUpwardThrust;
        } else {
            player.vy += shipGravity;
        }
        player.vy = Math.max(-maxVerticalSpeed, Math.min(maxVerticalSpeed, player.vy));

        // Pitch ship in the direction of flight
        player.angle = player.vy * 0.07;

        // Thrust flame particles
        trailParticles.push({
            x: player.x,
            y: player.y + player.size / 2 + (Math.random() - 0.5) * 6,
            vx: -3.5 - Math.random() * 2,
            vy: (Math.random() - 0.5) * 1.5,
            size: Math.random() * 5 + 2,
            life: 1.0,
            color: inputHeld ? '#ffaa00' : '#b537f2'
        });
    }

    player.x += player.vx;
    player.y += player.vy;

    // Update Moving Obstacles
    for (let mo of movingObstacles) {
        mo.y += mo.speed * mo.dir;
        if (mo.y > mo.baseY + mo.range) mo.dir = -1;
        if (mo.y < mo.baseY - mo.range) mo.dir = 1;

        // Moving obstacle collision
        if (player.x < mo.x + mo.size && player.x + player.size > mo.x &&
            player.y < mo.y + mo.size && player.y + player.size > mo.y) {
            die(); return;
        }
    }

    // Portal Touch Detection
    for (let p of portals) {
        if (player.x < p.x + 30 && player.x + player.size > p.x &&
            player.y < p.y + 70 && player.y + player.size > p.y) {
            if (player.mode !== p.toMode) {
                player.mode = p.toMode;
                // Burst effect
                for (let i = 0; i < 24; i++) {
                    portalBurstParticles.push({
                        x: p.x + 15, y: p.y + 35,
                        vx: (Math.random() - 0.5) * 8, vy: (Math.random() - 0.5) * 8,
                        size: Math.random() * 6 + 2, life: 1.0,
                        color: p.toMode === 'SHIP' ? '#ff00ea' : '#00f0ff'
                    });
                }
            }
        }
    }

    // Block Collisions
    player.onGround = false;
    for (let b of blocks) {
        if (player.x < b.x + b.w && player.x + player.size > b.x &&
            player.y < b.y + b.h && player.y + player.size > b.y) {
            if (player.mode === 'CUBE') {
                if (player.vy > 0 && player.y + player.size - player.vy <= b.y + 12) {
                    player.y = b.y - player.size;
                    player.vy = 0;
                    player.onGround = true;
                } else {
                    die(); return;
                }
            } else {
                // Ship mode: Touching blocks from any angle is fatal
                die(); return;
            }
        }
    }

    // Spike Collisions
    let pRect = { x: player.x + 5, y: player.y + 5, w: player.size - 10, h: player.size - 10 };
    for (let sp of spikes) {
        let sx = sp.x + 8, sy = sp.type === 'up' ? sp.y + 14 : sp.y, sw = 24, sh = 26;
        if (pRect.x < sx + sw && pRect.x + pRect.w > sx && pRect.y < sy + sh && pRect.y + pRect.h > sy) {
            die(); return;
        }
    }

    // Ceiling & Floor Screen Bounds for Ship
    if (player.y < 0 || player.y > canvas.height - player.size) {
        die(); return;
    }

    cameraX = player.x - 180;

    // Progress Calculation
    let passedCount = 0;
    for (let ox of obstacleXCoords) {
        if (player.x > ox) passedCount++;
        else break;
    }

    let rawPercent = Math.floor((passedCount / totalObstacles) * 100);
    currentProgress = Math.min(100, Math.max(0, rawPercent));
    
    uiProgressText.innerText = `PROGRESS: ${currentProgress}%`;
    uiProgressBar.style.width = `${currentProgress}%`;

    if (currentProgress > bestProgress) {
        bestProgress = currentProgress;
        localStorage.setItem('objectRunMap3Best', bestProgress);
        uiBest.innerText = `BEST: ${bestProgress}%`;
    }

    if (currentProgress >= 100) win();
}

function drawBackground(theme) {
    let grad = ctx.createLinearGradient(0, 0, 0, canvas.height);
    grad.addColorStop(0, theme.bg1);
    grad.addColorStop(1, theme.bg2);
    ctx.fillStyle = grad;
    ctx.fillRect(0, 0, canvas.width, canvas.height);

    // Subtle Speed Lines during Ship Flight
    if (player.mode === 'SHIP') {
        ctx.strokeStyle = 'rgba(255, 255, 255, 0.08)';
        ctx.lineWidth = 1.5;
        for (let i = 0; i < 6; i++) {
            let lineY = (i * 70 + (cameraX * 0.5)) % canvas.height;
            ctx.beginPath();
            ctx.moveTo(0, lineY);
            ctx.lineTo(canvas.width, lineY);
            ctx.stroke();
        }
    }

    // Grid Accent Lines
    ctx.strokeStyle = 'rgba(255, 255, 255, 0.04)';
    ctx.lineWidth = 1;
    ctx.beginPath();
    let offX = cameraX % TILE_SIZE;
    for (let i = 0; i <= canvas.width; i += TILE_SIZE) {
        ctx.moveTo(i - offX, 0); ctx.lineTo(i - offX, canvas.height);
    }
    for (let i = 0; i <= canvas.height; i += TILE_SIZE) {
        ctx.moveTo(0, i); ctx.lineTo(canvas.width, i);
    }
    ctx.stroke();
}

function draw() {
    const theme = getTheme(currentProgress);
    document.documentElement.style.setProperty('--theme-color', theme.main);
    document.documentElement.style.setProperty('--theme-glow', theme.glow);

    ctx.clearRect(0, 0, canvas.width, canvas.height);
    drawBackground(theme);

    ctx.save();
    ctx.translate(-cameraX, 0);

    // Render Transformation Portals
    for (let p of portals) {
        ctx.save();
        let portalGlow = p.toMode === 'SHIP' ? '#ff00ea' : '#00f0ff';
        ctx.shadowBlur = 18;
        ctx.shadowColor = portalGlow;
        ctx.strokeStyle = portalGlow;
        ctx.lineWidth = 4;
        
        // Oval Portal Ring
        ctx.beginPath();
        ctx.ellipse(p.x + 15, p.y + 40, 14, 45, 0, 0, Math.PI * 2);
        ctx.stroke();
        ctx.fillStyle = 'rgba(255, 255, 255, 0.2)';
        ctx.fill();
        ctx.restore();
    }

    // Render Trail / Exhaust Particles
    for (let i = trailParticles.length - 1; i >= 0; i--) {
        let t = trailParticles[i];
        t.x += t.vx || -1.5;
        t.y += t.vy || 0;
        t.life -= 0.035;
        if (t.life <= 0) {
            trailParticles.splice(i, 1);
            continue;
        }
        ctx.fillStyle = t.color;
        ctx.globalAlpha = Math.max(0, t.life);
        ctx.fillRect(t.x, t.y, t.size, t.size);
    }
    ctx.globalAlpha = 1.0;

    // Render Portal Burst Particles
    for (let i = portalBurstParticles.length - 1; i >= 0; i--) {
        let pb = portalBurstParticles[i];
        pb.x += pb.vx; pb.y += pb.vy; pb.life -= 0.03;
        if (pb.life <= 0) { portalBurstParticles.splice(i, 1); continue; }
        ctx.fillStyle = pb.color;
        ctx.globalAlpha = Math.max(0, pb.life);
        ctx.fillRect(pb.x, pb.y, pb.size, pb.size);
    }
    ctx.globalAlpha = 1.0;

    // Render Blocks
    ctx.shadowBlur = 10;
    ctx.shadowColor = theme.main;
    ctx.strokeStyle = theme.main;
    ctx.fillStyle = '#0a0a18';
    ctx.lineWidth = 2;
    for (let b of blocks) {
        if (b.x > cameraX - TILE_SIZE && b.x < cameraX + 850) {
            ctx.fillRect(b.x, b.y, b.w, b.h);
            ctx.strokeRect(b.x, b.y, b.w, b.h);
        }
    }

    // Render Moving Obstacles
    for (let mo of movingObstacles) {
        if (mo.x > cameraX - TILE_SIZE && mo.x < cameraX + 850) {
            ctx.shadowBlur = 14;
            ctx.shadowColor = '#ffaa00';
            ctx.fillStyle = '#ffaa00';
            ctx.fillRect(mo.x, mo.y, mo.size, mo.size);
            ctx.strokeStyle = '#fff';
            ctx.strokeRect(mo.x + 3, mo.y + 3, mo.size - 6, mo.size - 6);
        }
    }

    // Render Spikes (Floor & Ceiling)
    ctx.shadowColor = '#ff0055';
    ctx.fillStyle = '#ff0055';
    for (let sp of spikes) {
        if (sp.x > cameraX - TILE_SIZE && sp.x < cameraX + 850) {
            ctx.beginPath();
            if (sp.type === 'up') {
                ctx.moveTo(sp.x + TILE_SIZE / 2, sp.y + 4);
                ctx.lineTo(sp.x + TILE_SIZE - 4, sp.y + TILE_SIZE);
                ctx.lineTo(sp.x + 4, sp.y + TILE_SIZE);
            } else {
                ctx.moveTo(sp.x + 4, sp.y);
                ctx.lineTo(sp.x + TILE_SIZE - 4, sp.y);
                ctx.lineTo(sp.x + TILE_SIZE / 2, sp.y + TILE_SIZE - 4);
            }
            ctx.closePath();
            ctx.fill();
        }
    }

    // Render Player (CUBE vs. SHIP TRIANGLE)
    if (gameState !== 'GAMEOVER') {
        ctx.save();
        ctx.translate(player.x + player.size / 2, player.y + player.size / 2);
        ctx.rotate(player.angle);
        ctx.shadowBlur = 15;

        if (player.mode === 'CUBE') {
            ctx.shadowColor = '#00f0ff';
            ctx.fillStyle = '#00f0ff';
            ctx.fillRect(-player.size / 2, -player.size / 2, player.size, player.size);
            ctx.fillStyle = '#ffffff';
            ctx.fillRect(-player.size / 4, -player.size / 4, player.size / 2, player.size / 2);
        } else {
            // Triangle Flying Ship Shape
            ctx.shadowColor = '#b537f2';
            ctx.fillStyle = '#b537f2';
            ctx.beginPath();
            ctx.moveTo(player.size / 2 + 4, 0);                 // Ship Nose
            ctx.lineTo(-player.size / 2, -player.size / 2);     // Top Wing
            ctx.lineTo(-player.size / 3, 0);                    // Thruster Indent
            ctx.lineTo(-player.size / 2, player.size / 2);      // Bottom Wing
            ctx.closePath();
            ctx.fill();

            // Cockpit Window
            ctx.fillStyle = '#00f0ff';
            ctx.beginPath();
            ctx.arc(0, 0, 4.5, 0, Math.PI * 2);
            ctx.fill();
        }
        ctx.restore();
    }

    // Render Death Scatter Pieces
    if (gameState === 'GAMEOVER') {
        ctx.shadowBlur = 8;
        for (let p of particles) {
            if (p.life > 0) {
                p.vy += 0.55; p.x += p.vx; p.y += p.vy;
                p.rot += p.rotSpeed; p.life -= 0.022;
                ctx.save();
                ctx.translate(p.x, p.y);
                ctx.rotate(p.rot);
                ctx.globalAlpha = Math.max(0, p.life);
                ctx.shadowColor = p.color; ctx.fillStyle = p.color;
                ctx.fillRect(-p.size / 2, -p.size / 2, p.size, p.size);
                ctx.restore();
            }
        }
        ctx.globalAlpha = 1.0;
    }

    ctx.restore();

    if (gameState === 'START') {
        ctx.fillStyle = 'rgba(0, 0, 0, 0.6)';
        ctx.fillRect(0, 0, 800, 440);
        ctx.fillStyle = theme.main;
        ctx.font = '22px Orbitron';
        ctx.textAlign = 'center';
        ctx.shadowBlur = 12; ctx.shadowColor = theme.main;
        ctx.fillText("SPACE / CLICK: JUMP & FLY SHIP", 400, 220);
        ctx.textAlign = 'left';
    }
}

function loop() {
    update();
    draw();
    requestAnimationFrame(loop);
}
requestAnimationFrame(loop);
